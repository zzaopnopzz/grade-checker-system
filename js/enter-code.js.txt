async function searchStudents(query) {
  if (!query || query.length < 2) {
    hideStudentList();
    return;
  }

  try {
    const apiUrl = CONFIG.API[`M${grade}`];
    const url = `${apiUrl}?action=searchStudents&grade=${grade}&room=${room}&subject=${subject}&q=${encodeURIComponent(query)}`;
    
    const response = await fetch(url);
    const data = await response.json();

    if (data.success && data.students && data.students.length > 0) {
      displayStudentList(data.students);
    } else {
      hideStudentList();
      selectedStudent = null; // ← เคลียร์การเลือก
    }
  } catch (error) {
    console.error('Search error:', error);
    hideStudentList();
    selectedStudent = null; // ← เคลียร์การเลือก
  }
}

// ฟังก์ชันแสดงรายชื่อ
function displayStudentList(students) {
  const listDiv = document.getElementById('studentList');
  listDiv.innerHTML = '';
  
  students.forEach(student => {
    const item = document.createElement('div');
    item.className = 'student-item';
    item.innerHTML = `
      <div class="student-code">${student.code}</div>
      <div class="student-name">${student.name}</div>
      <div class="student-number">เลขที่ ${student.number}</div>
    `;
    
    // เมื่อคลิกเลือกนักเรียน
    item.addEventListener('click', () => {
      selectedStudent = student; // ← บันทึกข้อมูลนักเรียนที่เลือก
      document.getElementById('studentCode').value = student.code;
      document.getElementById('studentName').value = student.name;
      hideStudentList();
    });
    
    listDiv.appendChild(item);
  });
  
  listDiv.style.display = 'block';
}

// ฟังก์ชันซ่อนรายชื่อ
function hideStudentList() {
  document.getElementById('studentList').style.display = 'none';
}

// เมื่อพิมพ์ในช่องรหัส
document.getElementById('studentCode').addEventListener('input', (e) => {
  selectedStudent = null; // ← เคลียร์การเลือกเมื่อพิมพ์ใหม่
  searchStudents(e.target.value);
});

// เมื่อพิมพ์ในช่องชื่อ
document.getElementById('studentName').addEventListener('input', (e) => {
  selectedStudent = null; // ← เคลียร์การเลือกเมื่อพิมพ์ใหม่
  searchStudents(e.target.value);
});

// เมื่อกดปุ่มดูคะแนน
document.getElementById('submitBtn').addEventListener('click', async () => {
  const code = document.getElementById('studentCode').value.trim();
  const name = document.getElementById('studentName').value.trim();

  // ตรวจสอบว่าต้องเลือกจากรายชื่อ
  if (!selectedStudent) {
    alert('⚠️ กรุณาเลือกนักเรียนจากรายชื่อที่แสดง\n\nไม่สามารถพิมพ์ชื่อหรือรหัสเองได้');
    return;
  }

  // ตรวจสอบว่ารหัสตรงกับที่เลือกหรือไม่
  if (code !== selectedStudent.code) {
    alert('⚠️ รหัสนักเรียนไม่ตรงกับที่เลือก\n\nกรุณาเลือกนักเรียนจากรายชื่อใหม่');
    selectedStudent = null;
    return;
  }

  // ตรวจสอบว่ามีรหัสหรือไม่
  if (!code) {
    alert('⚠️ กรุณาเลือกนักเรียนจากรายชื่อ');
    return;
  }

  // แสดง Loading
  document.getElementById('submitBtn').disabled = true;
  document.getElementById('submitBtn').textContent = 'กำลังโหลด...';

  try {
    const apiUrl = CONFIG.API[`M${grade}`];
    const url = `${apiUrl}?action=getStudentScore&grade=${grade}&room=${room}&code=${encodeURIComponent(code)}&subject=${subject}`;
    
    const response = await fetch(url);
    const data = await response.json();

    if (data.success) {
      // บันทึกข้อมูลและไปหน้าแสดงผล
      sessionStorage.setItem('studentData', JSON.stringify(data));
      window.location.href = `show-score.html?subject=${subject}&grade=${grade}&room=${room}`;
    } else {
      alert('❌ ไม่พบข้อมูลนักเรียน\n\n' + (data.message || 'กรุณาลองใหม่อีกครั้ง'));
      selectedStudent = null;
    }
  } catch (error) {
    console.error('Error:', error);
    alert('❌ เกิดข้อผิดพลาดในการโหลดข้อมูล\n\nกรุณาตรวจสอบการเชื่อมต่ออินเทอร์เน็ต');
  } finally {
    document.getElementById('submitBtn').disabled = false;
    document.getElementById('submitBtn').textContent = '🔍 ดูคะแนน';
  }
});

// ซ่อนรายชื่อเมื่อคลิกนอกพื้นที่
document.addEventListener('click', (e) => {
  if (!e.target.closest('.search-container')) {
    hideStudentList();
  }
});

// ป้องกันการ Submit ด้วย Enter
document.getElementById('studentCode').addEventListener('keypress', (e) => {
  if (e.key === 'Enter') {
    e.preventDefault();
    if (selectedStudent) {
      document.getElementById('submitBtn').click();
    } else {
      alert('⚠️ กรุณาเลือกนักเรียนจากรายชื่อก่อน');
    }
  }
});
